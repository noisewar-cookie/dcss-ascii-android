# DCSS Android Font Configuration
# ──────────────────────────────────────────────────────────────────────────────
# Scale values are multipliers applied after the font is auto-sized to fill
# the panel width. 1.0 = auto-size only, 1.25 = 25% larger, 0.9 = 10% smaller.
#
# The col/row counts in comments are the terminal character dimensions of each
# panel region — they are fixed in the layout and shown here for reference only.
#
# *_scrollable=true  enables horizontal drag-scrolling within that panel/menu
#                    when the rendered 80-col bitmap is wider than the screen.
# *_vscrollable=true enables vertical drag-scrolling when the rendered bitmap
#                    is taller than the screen.
# Below the overflow threshold, the corresponding scrollable flag is a no-op.
# Drag area = the full rendered region of that panel/menu.
# ──────────────────────────────────────────────────────────────────────────────

# Default Game Screens
# Fallback for any in-game menu/info screen NOT matched by the more specific
# groups below. Includes: help screens (?), description popups (item/monster
# examine, lookup help ?/), message history (Ctrl+P), macros (~), and any
# other menu without an explicit category.
# Terminal region: 80 cols x 24 rows
portrait_default_font_scale=1.4
portrait_default_scrollable=true
portrait_default_vscrollable=true

# Portrait: pre-game loading screen
# Shown during: the static loading screen (text from string-array.xml,
# starting with "Launching game. May take a while the first time.")
# Defaults: falls back to portrait_default_* if unset
portrait_pregame_font_scale=1.5
portrait_pregame_scrollable=true

# Portrait: DCSS main menu
# Shown during: "Hello, welcome to Dungeon Crawl ..."
# Defaults: falls back to portrait_default_* if unset
portrait_mainmenu_font_scale=1.6
portrait_mainmenu_scrollable=true

# Portrait: map panel
# Shown during: gameplay — top portion, dungeon viewport
# Terminal region: 37 cols x 17 rows  (terminal cols 0-36, rows 0-16)
portrait_map_font_scale=0.85

# Portrait: map horizontal shift in terminal columns (positive = right)
# 0 = left-aligned, 2 = shifted 2 cols right (roughly centers the dungeon view)
portrait_map_offset_cols=4

# Portrait: HUD panel
# Shown during: gameplay — top portion, health/stats (right side of terminal)
# Terminal region: 43 cols x 17 rows  (terminal cols 37-79, rows 0-16)
portrait_hud_font_scale=0.85

# Portrait: HUD horizontal shift in terminal columns (positive = right)
# 0 = left-aligned, 1 aligns with the underscore at col 0 of the message panel
portrait_hud_offset_cols=1

# Portrait: message panel
# Shown during: gameplay — bottom strip, game messages
# Terminal region: 80 cols x 7 rows  (terminal cols 0-79, rows 17-23)
portrait_msg_font_scale=1.5

# Landscape: full terminal
# Shown during: all screens in landscape orientation
# Terminal region: 80 cols x 24 rows
landscape_font_scale=1.0

# ──────────────────────────────────────────────────────────────────────────────
# Per-menu overrides (portrait only). Each group below overrides
# portrait_default_font_scale / portrait_default_scrollable when that menu is
# shown.
# ──────────────────────────────────────────────────────────────────────────────

# Item lists: inventory (i), item-use prompts (w/W/T/P/R/q/r/v/f/t),
# drop (d), pickup (g), shop, stash search, known items (\),
# inscribe ({), quiver (Shift+Q)
# - portrait_items_scrollable: enable horizontal drag (when content > width)
# - portrait_items_vscrollable: enable vertical drag (when content > height)
portrait_items_font_scale=1.5
portrait_items_scrollable=true
portrait_items_vscrollable=true

# Spell/ability lists: spell list (I), cast (z), memorise (M),
# abilities (a), mutations (A)
portrait_spells_font_scale=1.5
portrait_spells_scrollable=true
portrait_spells_vscrollable=true

# Character overview (%)
portrait_overview_font_scale=1.6
portrait_overview_scrollable=true

# Skill training (m)
# In portrait, skill training is reorganized into a single column: column 0
# (Fighting...Stealth) is followed by column 1 (Spellcasting...Shapeshifting)
# without repeating the "Skill" header, then the help text. This makes the
# screen taller; vertical scrolling is needed at most font scales.
# - portrait_skills_scrollable: enable horizontal drag (when content > width)
# - portrait_skills_vscrollable: enable vertical drag (when content > height)
portrait_skills_font_scale=1.5
portrait_skills_scrollable=true
portrait_skills_vscrollable=false

# Religion screen (^)
portrait_religion_font_scale=1.4
portrait_religion_scrollable=true

# Hi-score screen (end of game)
portrait_hiscores_font_scale=1.4
portrait_hiscores_scrollable=true

# Stash search results (Ctrl+F → match found)
# Shown during: the menu listing matching items/features after a search.
# Detected by the "[toggle:" marker in the title row.
portrait_travel_font_scale=1.5
portrait_travel_scrollable=true
portrait_travel_vscrollable=true

# Dungeon level map (Shift+X)
# Shown during: the level overview / travel map. Detected by the
# "(Press ? for help)" marker drawn at the right of the title row.
portrait_levelmap_font_scale=1.75
portrait_levelmap_scrollable=true
portrait_levelmap_vscrollable=true

# Visible features (Ctrl+X)
# Shown during: the "Visible Monsters/Items/Features" target/travel menu.
# Detected by the "Visible " prefix at the start of the title row.
portrait_vfeatures_font_scale=1.5
portrait_vfeatures_scrollable=true
portrait_vfeatures_vscrollable=true

# ──────────────────────────────────────────────────────────────────────────────
# Newgame species/background screens (portrait)
# Each category renders into its own vertically-stacked panel that samples a
# fixed terminal rectangle from the upstream multi-column layout. Categories
# you don't play with (e.g. Mage if you only ever pick warriors) can be left
# alone; their panels still render but you won't notice the scale.
# Detection: row 0 contains "Welcome" + "species." or "background.".
# ──────────────────────────────────────────────────────────────────────────────

# Each newgame panel has its own scrollable/vscrollable flag. Same semantics
# as elsewhere: scrollable enables horizontal drag when content is wider than
# the panel, vscrollable enables vertical drag when content is taller.
# Defaults: portrait_default_scrollable / portrait_default_vscrollable.

# Welcome row at the top of the screen ("Welcome, %s. Please select your ...")
# Terminal region: 80 cols x 2 rows  (rows 0-1)
portrait_newgame_welcome_font_scale=1.4
portrait_newgame_welcome_scrollable=true
portrait_newgame_welcome_vscrollable=false

# All species/background category panels share one knob set. Each samples a
# sub-slice of the upstream multi-column grid, but the font is sized as if
# rendering the full 80-col terminal — so a multiplier of 1.5 here matches
# the mainmenu glyph size.
portrait_newgame_category_font_scale=1.5
portrait_newgame_category_scrollable=true
portrait_newgame_category_vscrollable=false

# Description text shown above the sub-options (selected species/background
# description). Rendered into its own full-width panel that samples the
# rows between the category grid and the sub-options grid. Often wider than
# the screen, so horizontal scrolling is typically wanted.
portrait_newgame_desc_font_scale=1.4
portrait_newgame_desc_scrollable=true
portrait_newgame_desc_vscrollable=false

# Sub-options block at the bottom (random/aptitudes/help/space/tab). The
# upstream 2-col grid is split into two stacked panels (left col then right
# col) so the 8 prompts read as a single vertical list.
portrait_newgame_sub_font_scale=1.4
portrait_newgame_sub_scrollable=true
portrait_newgame_sub_vscrollable=false
