# DCSS Android Font Configuration
# ──────────────────────────────────────────────────────────────────────────────
# Sizing model: every user-selectable font is auto-sized using VeraMoBd as
# a reference — the auto-sizer picks a textSize such that VeraMoBd would fit
# the panel width, then applies that same textSize to the chosen font. Line
# height stays consistent across faces; widths vary with each font's natural
# aspect (wider fonts like Flexi_IBM_VGA_False overflow symmetrically past
# the panel edges, narrower ones like Flexi_IBM_VGA_True leave slack).
#
# There is no horizontal scaleX distortion; there are no aspect-normalization
# knobs.
# ──────────────────────────────────────────────────────────────────────────────

# ──────────────────────────────────────────────────────────────────────────────
# Per-panel scale values are multipliers applied after the font is auto-sized
# to fill the panel width. 1.0 = auto-size only, 1.25 = 25% larger, 0.9 = 10%
# smaller.
#
# The col/row counts in comments are the terminal character dimensions of each
# panel region — they are fixed in the layout and shown here for reference only.
#
# *_scrollable=true  enables horizontal drag-scrolling within that panel/menu
#                    when the rendered 80-col bitmap is wider than the screen.
# *_vscrollable=true enables vertical drag-scrolling when the rendered bitmap
#                    is taller than the screen.
# Below the overflow threshold, the corresponding scrollable flag is a no-op.
# Drag area = the full rendered region of that panel/menu.
# ──────────────────────────────────────────────────────────────────────────────

# Portrait: map horizontal nudge in terminal columns (positive = right)

# Portrait: map vertical-slack behaviour when the map is width-constrained
# (the usual portrait case, where the 33-col dungeon view fills the width
# before it fills the height). false = keep the width-fit block, bottom-
# pinned to the HUD with fit slack above it. true = grow the font to fill
# the panel height (bigger glyphs; left/right columns crop, clipped off the
# bars).

# Portrait: HUD panel (stats only; monster list is a separate panel below

portrait_hud_font_scale=0.85

# Portrait: HUD horizontal shift in terminal columns (positive = right)
# 0 = left-aligned (Edge Margin pref handles global inset now)
portrait_hud_offset_cols=0

# Compact HUD: vertical HP/MP bar glyphs (Unicode codepoint each). HP and MP
# are set independently — each has its own filled-cell and empty-track glyph.
# Accepts U+XXXX / 0xXXXX / #XXXX (hex) or a plain decimal. Pick codepoints the
# game font actually has (CP437 block/box glyphs), else they render as tofu.
# fill examples:  U+2592 ▒ medium, U+2593 ▓ dark, U+2588 █ full block.
# empty examples: U+007C | pipe, U+00B7 · middot, U+2591 ░ light shade.
compact_hp_bar_glyph=U+2588
compact_hp_bar_empty_glyph=U+007C
compact_mp_bar_glyph=U+2588
compact_mp_bar_empty_glyph=U+007C

# Compact HUD: vertical MP bar fill colour (RRGGBB / #RRGGBB / 0xRRGGBB /
# 8-digit ARGB). Overrides the console MP bar BLUE fill. Default 305FE8 = DCSS
# BLUE (0040FF) with HSL lightness +10% and saturation -20%.
compact_mp_bar_color=305FE8

# Standard HUD: horizontal MP bar fill colour (same formats). Recolours the
# terminal MP bar to match the compact vertical bar above. Defaults to
# compact_mp_bar_color, so set that alone to move both together, or override
# here to colour the two bars independently.
mp_bar_color=305FE8

# Dungeon level map (Shift+X)
# Shown during: the level overview / travel map. Detected by the
# "(Press ? for help)" marker drawn at the right of the title row.
portrait_levelmap_font_scale=2.25
portrait_levelmap_scrollable=true
portrait_levelmap_vscrollable=true
# Pinch-zoom max is an absolute multiplier of the base LEVELMAP scale
# (e.g. 1.5 = 1.5x zoom in). Session-only: contentZoom resets to 1.0
# whenever the map screen closes. The min zoom-out is computed
# dynamically from viewport / bitmap size so the full 80×48-cell map
# always fits at max zoom-out — no manual knob to keep in sync.
portrait_levelmap_pinch_zoom_max=1.5
# Sensitivity = exponent applied per gesture event to the raw pinch factor.
# 1.0 = raw system response; <1.0 = less sensitive (more finger motion for
# the same zoom step); >1.0 = more sensitive. Zoom-in and zoom-out are
# separate because pinch-out (fingers spreading) and pinch-in (fingers
# closing) can feel asymmetric on real hands.
portrait_levelmap_pinch_zoom_in_sensitivity=1.0
portrait_levelmap_pinch_zoom_out_sensitivity=0.75
